The response body streams as plain text and resolves to a JSON object:

  "tracks": {
    "bass": "36:0.0:0.5:8|36:0.5:0.5:6|38:1.0:0.5:7|...",
    "mel":  "60:0.0:1.0:7|62:1.0:0.5:6|...",
    "pad":  "48:0.0:8.0:3|52:0.0:8.0:2|..."
  }
}

Note Data Format

Each track value is a |-separated string of pitch:beat:duration:velocity:

  • pitch: MIDI note number (0–127)
  • beat: Start time in beats (float, multiples of 0.25)
  • duration: Length in beats (float)
  • velocity: 1–9 (mapped internally to 1–127)

Binary MIDI (?format=midi)

Returns audio/midi with Content-Disposition: attachment. Ready to drag into Ableton, FL Studio, Logic, etc.

💻 Examples

JavaScript (Streaming JSON)

const res = await fetch('https://midi-aigenerator.vercel.app/api/v1/midi', {
  method: 'POST',
  headers: { 'Authorization': 'Bearer mgk_YOUR_KEY', 'Content-Type': 'application/json' },
  body: JSON.stringify({ promptParams: { bpm: 130, style: 'techno', bars: 8, trackTypes: ['bassline','lead','perc'] } })
});

const reader = res.body.getReader();
const decoder = new TextDecoder();
let text = '';
while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  text += decoder.decode(value);
}
const midi = JSON.parse(text);
console.log(midi.title, Object.keys(midi.tracks));

JavaScript (Download .mid)

const res = await fetch('https://midi-aigenerator.vercel.app/api/v1/midi?format=midi', {
  method: 'POST',
  headers: { 'Authorization': 'Bearer mgk_YOUR_KEY', 'Content-Type': 'application/json' },
  body: JSON.stringify({ promptParams: { key: 'F', scale: 'minor', bpm: 120, bars: 4 } })
});
const blob = await res.blob();
const url = URL.createObjectURL(blob);
const a = document.createElement('a');
a.href = url; a.download = 'my-track.mid'; a.click();

Python

import requests, json

# JSON Stream
r = requests.post(
  'https://midi-aigenerator.vercel.app/api/v1/midi',
  headers={'Authorization': 'Bearer mgk_YOUR_KEY', 'Content-Type': 'application/json'},
  json={'promptParams': {'bpm': 110, 'style': 'jazz', 'trackTypes': ['bassline','keys','melody']}},
  stream=True
)
midi = json.loads(''.join(c.decode() for c in r.iter_content(chunk_size=None)))
print(midi['title'])

# Save .mid
r = requests.post(
  'https://midi-aigenerator.vercel.app/api/v1/midi?format=midi',
  headers={'Authorization': 'Bearer mgk_YOUR_KEY', 'Content-Type': 'application/json'},
  json={'promptParams': {'bpm': 110, 'style': 'jazz', 'trackTypes': ['bassline','keys','melody']}}
)
with open('track.mid', 'wb') as f: f.write(r.content)
